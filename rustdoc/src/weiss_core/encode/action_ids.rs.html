8/// Parameter value for an action id description.
9#[derive(Clone, Debug, Serialize, PartialEq, Eq)]
10#[serde(untagged)]
11pub enum ActionParamValue {
12    /// Integer parameter.
13    Int(i32),
14    /// String parameter.
15    Str(&'static str),
16}
17
18/// Named parameter for an action id description.
19#[derive(Clone, Debug, Serialize, PartialEq, Eq)]
20pub struct ActionParam {
21    /// Parameter name.
22    pub name: &'static str,
23    /// Parameter value.
24    pub value: ActionParamValue,
25}

27/// Human-readable description of an action id.
28#[derive(Clone, Debug, Serialize, PartialEq, Eq)]
29pub struct ActionIdDesc {
30    /// Action family name.
31    pub family: &'static str,
32    /// Parameters associated with the action.
33    pub params: Vec<ActionParam>,
34}

36/// Decode an action id into a human-readable description.
37pub fn decode_action_id(id: usize) -> Option<ActionIdDesc> {
38    if id >= ACTION_SPACE_SIZE {
39        return None;
40    }
41    if id == MULLIGAN_CONFIRM_ID {
42        return Some(ActionIdDesc {
43            family: "mulligan_confirm",
44            params: vec![],
45        });
46    }
47    if (MULLIGAN_SELECT_BASE..MULLIGAN_SELECT_BASE + MULLIGAN_SELECT_COUNT).contains(&id) {
48        let hand_index = (id - MULLIGAN_SELECT_BASE) as i32;
49        return Some(ActionIdDesc {
50            family: "mulligan_select",
51            params: vec![ActionParam {
52                name: "hand_index",
53                value: ActionParamValue::Int(hand_index),
54            }],
55        });
56    }
57    if id == PASS_ACTION_ID {
58        return Some(ActionIdDesc {
59            family: "pass",
60            params: vec![],
61        });
62    }
63    if (CLOCK_HAND_BASE..CLOCK_HAND_BASE + CLOCK_HAND_COUNT).contains(&id) {
64        let hand_index = (id - CLOCK_HAND_BASE) as i32;
65        return Some(ActionIdDesc {
66            family: "clock_from_hand",
67            params: vec![ActionParam {
68                name: "hand_index",
69                value: ActionParamValue::Int(hand_index),
70            }],
71        });
72    }
73    if (MAIN_PLAY_CHAR_BASE..MAIN_PLAY_CHAR_BASE + MAIN_PLAY_CHAR_COUNT).contains(&id) {
74        let offset = id - MAIN_PLAY_CHAR_BASE;
75        let hand_index = (offset / MAX_STAGE) as i32;
76        let stage_slot = (offset % MAX_STAGE) as i32;
77        return Some(ActionIdDesc {
78            family: "main_play_character",
79            params: vec![
80                ActionParam {
81                    name: "hand_index",
82                    value: ActionParamValue::Int(hand_index),
83                },
84                ActionParam {
85                    name: "stage_slot",
86                    value: ActionParamValue::Int(stage_slot),
87                },
88            ],
89        });
90    }
91    if (MAIN_PLAY_EVENT_BASE..MAIN_PLAY_EVENT_BASE + MAIN_PLAY_EVENT_COUNT).contains(&id) {
92        let hand_index = (id - MAIN_PLAY_EVENT_BASE) as i32;
93        return Some(ActionIdDesc {
94            family: "main_play_event",
95            params: vec![ActionParam {
96                name: "hand_index",
97                value: ActionParamValue::Int(hand_index),
98            }],
99        });
100    }
101    if (MAIN_MOVE_BASE..MAIN_MOVE_BASE + MAIN_MOVE_COUNT).contains(&id) {
102        let offset = id - MAIN_MOVE_BASE;
103        let from_slot = offset / (MAX_STAGE - 1);
104        let to_index = offset % (MAX_STAGE - 1);
105        let to_slot = if to_index >= from_slot {
106            to_index + 1
107        } else {
108            to_index
109        };
110        return Some(ActionIdDesc {
111            family: "main_move",
112            params: vec![
113                ActionParam {
114                    name: "from_slot",
115                    value: ActionParamValue::Int(from_slot as i32),
116                },
117                ActionParam {
118                    name: "to_slot",
119                    value: ActionParamValue::Int(to_slot as i32),
120                },
121            ],
122        });
123    }
124    if (CLIMAX_PLAY_BASE..CLIMAX_PLAY_BASE + CLIMAX_PLAY_COUNT).contains(&id) {
125        let hand_index = (id - CLIMAX_PLAY_BASE) as i32;
126        return Some(ActionIdDesc {
127            family: "climax_play",
128            params: vec![ActionParam {
129                name: "hand_index",
130                value: ActionParamValue::Int(hand_index),
131            }],
132        });
133    }
134    if (ATTACK_BASE..ATTACK_BASE + ATTACK_COUNT).contains(&id) {
135        let offset = id - ATTACK_BASE;
136        let slot = (offset / 3) as i32;
137        let attack_type = match (offset % 3) as i32 {
138            0 => "frontal",
139            1 => "side",
140            _ => "direct",
141        };
142        return Some(ActionIdDesc {
143            family: "attack",
144            params: vec![
145                ActionParam {
146                    name: "slot",
147                    value: ActionParamValue::Int(slot),
148                },
149                ActionParam {
150                    name: "attack_type",
151                    value: ActionParamValue::Str(attack_type),
152                },
153            ],
154        });
155    }
156    if (LEVEL_UP_BASE..LEVEL_UP_BASE + LEVEL_UP_COUNT).contains(&id) {
157        let index = (id - LEVEL_UP_BASE) as i32;
158        return Some(ActionIdDesc {
159            family: "level_up",
160            params: vec![ActionParam {
161                name: "index",
162                value: ActionParamValue::Int(index),
163            }],
164        });
165    }
166    if (ENCORE_PAY_BASE..ENCORE_PAY_BASE + ENCORE_PAY_COUNT).contains(&id) {
167        let slot = (id - ENCORE_PAY_BASE) as i32;
168        return Some(ActionIdDesc {
169            family: "encore_pay",
170            params: vec![ActionParam {
171                name: "slot",
172                value: ActionParamValue::Int(slot),
173            }],
174        });
175    }
176    if (ENCORE_DECLINE_BASE..ENCORE_DECLINE_BASE + ENCORE_DECLINE_COUNT).contains(&id) {
177        let slot = (id - ENCORE_DECLINE_BASE) as i32;
178        return Some(ActionIdDesc {
179            family: "encore_decline",
180            params: vec![ActionParam {
181                name: "slot",
182                value: ActionParamValue::Int(slot),
183            }],
184        });
185    }
186    if (TRIGGER_ORDER_BASE..TRIGGER_ORDER_BASE + TRIGGER_ORDER_COUNT).contains(&id) {
187        let index = (id - TRIGGER_ORDER_BASE) as i32;
188        return Some(ActionIdDesc {
189            family: "trigger_order",
190            params: vec![ActionParam {
191                name: "index",
192                value: ActionParamValue::Int(index),
193            }],
194        });
195    }
196    if (CHOICE_BASE..CHOICE_BASE + CHOICE_COUNT).contains(&id) {
197        let index = (id - CHOICE_BASE) as i32;
198        return Some(ActionIdDesc {
199            family: "choice_select",
200            params: vec![ActionParam {
201                name: "index",
202                value: ActionParamValue::Int(index),
203            }],
204        });
205    }
206    if id == CHOICE_PREV_ID {
207        return Some(ActionIdDesc {
208            family: "choice_prev_page",
209            params: vec![],
210        });
211    }
212    if id == CHOICE_NEXT_ID {
213        return Some(ActionIdDesc {
214            family: "choice_next_page",
215            params: vec![],
216        });
217    }
218    if id == CONCEDE_ID {
219        return Some(ActionIdDesc {
220            family: "concede",
221            params: vec![],
222        });
223    }
224    None
225}

227/// Decode an action id into a canonical action descriptor.
228pub fn action_desc_for_id(id: usize) -> Option<ActionDesc> {
229    if id >= ACTION_SPACE_SIZE {
230        return None;
231    }
232    if id == MULLIGAN_CONFIRM_ID {
233        return Some(ActionDesc::MulliganConfirm);
234    }
235    if (MULLIGAN_SELECT_BASE..MULLIGAN_SELECT_BASE + MULLIGAN_SELECT_COUNT).contains(&id) {
236        let hand_index = (id - MULLIGAN_SELECT_BASE) as u8;
237        return Some(ActionDesc::MulliganSelect { hand_index });
238    }
239    if id == PASS_ACTION_ID {
240        return Some(ActionDesc::Pass);
241    }
242    if (CLOCK_HAND_BASE..CLOCK_HAND_BASE + CLOCK_HAND_COUNT).contains(&id) {
243        let hand_index = (id - CLOCK_HAND_BASE) as u8;
244        return Some(ActionDesc::Clock { hand_index });
245    }
246    if (MAIN_PLAY_CHAR_BASE..MAIN_PLAY_CHAR_BASE + MAIN_PLAY_CHAR_COUNT).contains(&id) {
247        let offset = id - MAIN_PLAY_CHAR_BASE;
248        let hand_index = (offset / MAX_STAGE) as u8;
249        let stage_slot = (offset % MAX_STAGE) as u8;
250        return Some(ActionDesc::MainPlayCharacter {
251            hand_index,
252            stage_slot,
253        });
254    }
255    if (MAIN_PLAY_EVENT_BASE..MAIN_PLAY_EVENT_BASE + MAIN_PLAY_EVENT_COUNT).contains(&id) {
256        let hand_index = (id - MAIN_PLAY_EVENT_BASE) as u8;
257        return Some(ActionDesc::MainPlayEvent { hand_index });
258    }
259    if (MAIN_MOVE_BASE..MAIN_MOVE_BASE + MAIN_MOVE_COUNT).contains(&id) {
260        let offset = id - MAIN_MOVE_BASE;
261        let from_slot = (offset / (MAX_STAGE - 1)) as u8;
262        let to_index = (offset % (MAX_STAGE - 1)) as u8;
263        let to_slot = if to_index < from_slot {
264            to_index
265        } else {
266            to_index + 1
267        };
268        return Some(ActionDesc::MainMove { from_slot, to_slot });
269    }
270    if (CLIMAX_PLAY_BASE..CLIMAX_PLAY_BASE + CLIMAX_PLAY_COUNT).contains(&id) {
271        let hand_index = (id - CLIMAX_PLAY_BASE) as u8;
272        return Some(ActionDesc::ClimaxPlay { hand_index });
273    }
274    if (ATTACK_BASE..ATTACK_BASE + ATTACK_COUNT).contains(&id) {
275        let offset = id - ATTACK_BASE;
276        let slot = (offset / 3) as u8;
277        let attack_type = match (offset % 3) as i32 {
278            0 => AttackType::Frontal,
279            1 => AttackType::Side,
280            _ => AttackType::Direct,
281        };
282        return Some(ActionDesc::Attack { slot, attack_type });
283    }
284    if (LEVEL_UP_BASE..LEVEL_UP_BASE + LEVEL_UP_COUNT).contains(&id) {
285        let index = (id - LEVEL_UP_BASE) as u8;
286        return Some(ActionDesc::LevelUp { index });
287    }
288    if (ENCORE_PAY_BASE..ENCORE_PAY_BASE + ENCORE_PAY_COUNT).contains(&id) {
289        let slot = (id - ENCORE_PAY_BASE) as u8;
290        return Some(ActionDesc::EncorePay { slot });
291    }
292    if (ENCORE_DECLINE_BASE..ENCORE_DECLINE_BASE + ENCORE_DECLINE_COUNT).contains(&id) {
293        let slot = (id - ENCORE_DECLINE_BASE) as u8;
294        return Some(ActionDesc::EncoreDecline { slot });
295    }
296    if (TRIGGER_ORDER_BASE..TRIGGER_ORDER_BASE + TRIGGER_ORDER_COUNT).contains(&id) {
297        let index = (id - TRIGGER_ORDER_BASE) as u8;
298        return Some(ActionDesc::TriggerOrder { index });
299    }
300    if (CHOICE_BASE..CHOICE_BASE + CHOICE_COUNT).contains(&id) {
301        let index = (id - CHOICE_BASE) as u8;
302        return Some(ActionDesc::ChoiceSelect { index });
303    }
304    if id == CHOICE_PREV_ID {
305        return Some(ActionDesc::ChoicePrevPage);
306    }
307    if id == CHOICE_NEXT_ID {
308        return Some(ActionDesc::ChoiceNextPage);
309    }
310    if id == CONCEDE_ID {
311        return Some(ActionDesc::Concede);
312    }
313    None
314}

316/// Encode a canonical action descriptor into an action id.
317pub fn action_id_for(action: &ActionDesc) -> Option<usize> {
318    match action {
319        ActionDesc::MulliganConfirm => Some(MULLIGAN_CONFIRM_ID),
320        ActionDesc::MulliganSelect { hand_index } => {
321            let hi = *hand_index as usize;
322            if hi < MULLIGAN_SELECT_COUNT {
323                Some(MULLIGAN_SELECT_BASE + hi)
324            } else {
325                None
326            }
327        }
328        ActionDesc::Pass => Some(PASS_ACTION_ID),
329        ActionDesc::Clock { hand_index } => {
330            let hi = *hand_index as usize;
331            if hi < MAX_HAND {
332                Some(CLOCK_HAND_BASE + hi)
333            } else {
334                None
335            }
336        }
337        ActionDesc::MainPlayCharacter {
338            hand_index,
339            stage_slot,
340        } => {
341            let hi = *hand_index as usize;
342            let ss = *stage_slot as usize;
343            if hi < MAX_HAND && ss < MAX_STAGE {
344                Some(MAIN_PLAY_CHAR_BASE + hi * MAX_STAGE + ss)
345            } else {
346                None
347            }
348        }
349        ActionDesc::MainPlayEvent { hand_index } => {
350            let hi = *hand_index as usize;
351            if hi < MAX_HAND {
352                Some(MAIN_PLAY_EVENT_BASE + hi)
353            } else {
354                None
355            }
356        }
357        ActionDesc::MainMove { from_slot, to_slot } => {
358            let fs = *from_slot as usize;
359            let ts = *to_slot as usize;
360            if fs < MAX_STAGE && ts < MAX_STAGE && fs != ts {
361                let to_index = if ts < fs { ts } else { ts - 1 };
362                Some(MAIN_MOVE_BASE + fs * (MAX_STAGE - 1) + to_index)
363            } else {
364                None
365            }
366        }
367        ActionDesc::MainActivateAbility { .. } => None,
368        ActionDesc::ClimaxPlay { hand_index } => {
369            let hi = *hand_index as usize;
370            if hi < MAX_HAND {
371                Some(CLIMAX_PLAY_BASE + hi)
372            } else {
373                None
374            }
375        }
376        ActionDesc::Attack { slot, attack_type } => {
377            let s = *slot as usize;
378            let t = attack_type_to_i32(*attack_type) as usize;
379            if s < ATTACK_SLOT_COUNT && t < 3 {
380                Some(ATTACK_BASE + s * 3 + t)
381            } else {
382                None
383            }
384        }
385        ActionDesc::CounterPlay { .. } => None,
386        ActionDesc::LevelUp { index } => {
387            let idx = *index as usize;
388            if idx < LEVEL_UP_COUNT {
389                Some(LEVEL_UP_BASE + idx)
390            } else {
391                None
392            }
393        }
394        ActionDesc::EncorePay { slot } => {
395            let s = *slot as usize;
396            if s < ENCORE_PAY_COUNT {
397                Some(ENCORE_PAY_BASE + s)
398            } else {
399                None
400            }
401        }
402        ActionDesc::EncoreDecline { slot } => {
403            let s = *slot as usize;
404            if s < ENCORE_DECLINE_COUNT {
405                Some(ENCORE_DECLINE_BASE + s)
406            } else {
407                None
408            }
409        }
410        ActionDesc::TriggerOrder { index } => {
411            let idx = *index as usize;
412            if idx < TRIGGER_ORDER_COUNT {
413                Some(TRIGGER_ORDER_BASE + idx)
414            } else {
415                None
416            }
417        }
418        ActionDesc::ChoiceSelect { index } => {
419            let idx = *index as usize;
420            if idx < CHOICE_COUNT {
421                Some(CHOICE_BASE + idx)
422            } else {
423                None
424            }
425        }
426        ActionDesc::ChoicePrevPage => Some(CHOICE_PREV_ID),
427        ActionDesc::ChoiceNextPage => Some(CHOICE_NEXT_ID),
428        ActionDesc::Concede => Some(CONCEDE_ID),
429    }
430}

432fn attack_type_to_i32(attack_type: AttackType) -> i32 {
433    match attack_type {
434        AttackType::Frontal => 0,
435        AttackType::Side => 1,
436        AttackType::Direct => 2,
437    }
