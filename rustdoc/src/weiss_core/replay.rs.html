159impl ReplayConfig {
160    /// Recompute cached sampling threshold after changing `sample_rate`.
161    pub fn rebuild_cache(&mut self) {
162        let rate = self.sample_rate.clamp(0.0, 1.0);
163        self.sample_threshold = if rate <= 0.0 {
164            0
165        } else if rate >= 1.0 {
166            u32::MAX
167        } else {
168            (rate * (u32::MAX as f32)).round() as u32
169        };
170    }
171}

173/// Background replay writer that serializes episodes to disk.
174#[derive(Clone)]
175pub struct ReplayWriter {
176    sender: Sender<ReplayData>,
177}
178
179impl ReplayWriter {
180    /// Spawn a background writer for the given config.
181    pub fn new(config: &ReplayConfig) -> Result<Self> {
182        fs::create_dir_all(&config.out_dir).context("Failed to create replay output directory")?;
183        let (tx, rx) = mpsc::channel::<ReplayData>();
184        let out_dir = config.out_dir.clone();
185        let compress = config.compress;
186        thread::spawn(move || {
187            for data in rx.into_iter() {
188                let header = &data.header;
189                let filename = format!(
190                    "episode_{:04}_{:08}_{:016x}.wsr",
191                    header.env_id, header.episode_index, header.seed
192                );
193                let path = out_dir.join(filename);
194                if let Err(err) = write_replay_file(&path, &data, compress) {
195                    eprintln!("Replay write failed: {err}");
196                }
197            }
198        });
199        Ok(Self { sender: tx })
200    }
201
202    /// Enqueue replay data for async write.
203    #[allow(clippy::result_large_err)]
204    pub fn send(&self, data: ReplayData) -> std::result::Result<(), mpsc::SendError<ReplayData>> {
205        self.sender.send(data)
206    }
207}

209fn write_replay_file(path: &Path, data: &ReplayData, compress: bool) -> Result<()> {
210    // Write to a sidecar temp file first and atomically rename into place so readers never
211    // observe partially-written replay payloads.
212    let mut tmp_path = path.to_path_buf();
213    let tmp_extension = path
214        .extension()
215        .map(|ext| format!("{}.tmp", ext.to_string_lossy()))
216        .unwrap_or_else(|| "tmp".to_string());
217    tmp_path.set_extension(tmp_extension);
218    let mut file = File::create(&tmp_path)?;
219    write_replay_to_writer(&mut file, data, compress)?;
220    file.flush()?;
221    file.sync_all()?;
222    fs::rename(&tmp_path, path)?;
223    Ok(())
224}

226fn write_replay_to_writer<W: Write>(
227    writer: &mut W,
228    data: &ReplayData,
229    compress: bool,
230) -> Result<()> {
231    let base = postcard::to_stdvec(data)?;
232    let payload = if compress {
233        #[cfg(feature = "replay-zstd")]
234        {
235            zstd::stream::encode_all(&base[..], 3)?
236        }
237        #[cfg(not(feature = "replay-zstd"))]
238        {
239            anyhow::bail!("Replay compression requested but replay-zstd feature is disabled");
240        }
241    } else {
242        base
243    };
244    let mut len_bytes = Vec::with_capacity(8);
245    let len_flag = write_payload_len(&mut len_bytes, payload.len())?;
246    let mut flags = if compress { FLAG_COMPRESSED } else { 0 };
247    flags |= len_flag;
248    writer.write_all(MAGIC)?;
249    writer.write_all(&[flags])?;
250    writer.write_all(&len_bytes)?;
251    writer.write_all(&payload)?;
252    Ok(())
253}
254
255fn write_payload_len<W: Write>(writer: &mut W, payload_len: usize) -> Result<u8> {
256    let len = u64::try_from(payload_len).context("Replay payload length exceeds u64 range")?;
257    if len > u64::from(u32::MAX) {
258        writer.write_all(&len.to_le_bytes())?;
259        Ok(FLAG_PAYLOAD_LEN_U64)
260    } else {
261        writer.write_all(&(len as u32).to_le_bytes())?;
262        Ok(0)
263    }
264}
265
266fn read_payload_len<R: Read>(reader: &mut R, flags: u8) -> Result<usize> {
267    let len = if (flags & FLAG_PAYLOAD_LEN_U64) != 0 {
268        let mut len_bytes = [0u8; 8];
269        reader.read_exact(&mut len_bytes)?;
270        u64::from_le_bytes(len_bytes)
271    } else {
272        let mut len_bytes = [0u8; 4];
273        reader.read_exact(&mut len_bytes)?;
274        u64::from(u32::from_le_bytes(len_bytes))
275    };
276    usize::try_from(len).context("Replay payload length exceeds platform limits")
277}

279/// Read and decode a replay file from disk.
280pub fn read_replay_file(path: &Path) -> Result<ReplayData> {
281    let mut file = File::open(path)?;
282    read_replay_from_reader(&mut file)
283}
284
285fn read_replay_from_reader<R: Read>(reader: &mut R) -> Result<ReplayData> {
286    let mut magic = [0u8; 4];
287    reader.read_exact(&mut magic)?;
288    if &magic != MAGIC {
289        anyhow::bail!("Invalid replay magic");
290    }
291    let mut flag = [0u8; 1];
292    reader.read_exact(&mut flag)?;
293    let flags = flag[0];
294    let len = read_payload_len(reader, flags)?;
295    let mut payload = vec![0u8; len];
296    reader.read_exact(&mut payload)?;
297    let compressed = (flags & FLAG_COMPRESSED) != 0;
298    if compressed {
299        #[cfg(feature = "replay-zstd")]
300        {
301            payload = zstd::stream::decode_all(&payload[..])?;
302        }
303        #[cfg(not(feature = "replay-zstd"))]
304        {
305            anyhow::bail!("Replay file is compressed but replay-zstd feature is disabled");
306        }
307    }
308    let data: ReplayData = postcard::from_bytes(&payload)?;
309    Ok(data)
310}
