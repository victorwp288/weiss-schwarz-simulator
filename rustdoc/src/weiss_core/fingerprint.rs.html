15/// Fingerprint algorithm identifier.
16pub const FINGERPRINT_ALGO: &str = "postcard+blake3+u64le v1";
17
18/// Hash raw bytes into a stable 64-bit fingerprint.
19pub fn hash_bytes(bytes: &[u8]) -> u64 {
20    let hash = blake3::hash(bytes);
21    let mut out = [0u8; 8];
22    out.copy_from_slice(&hash.as_bytes()[..8]);
23    u64::from_le_bytes(out)
24}
25
26/// Hash a serializable value using postcard + blake3.
27pub fn hash_postcard<T: Serialize + ?Sized>(value: &T) -> u64 {
28    let bytes = postcard::to_allocvec(value).expect("fingerprint serialization failed");
29    hash_bytes(&bytes)
30}

32/// Compute a stable fingerprint for env config + curriculum.
33pub fn config_fingerprint(config: &EnvConfig, curriculum: &CurriculumConfig) -> u64 {
34    let canonical = CanonicalConfigForHash::from_config(config, curriculum);
35    hash_postcard(&canonical)
36}

38/// Compute a stable fingerprint for a game state.
39pub fn state_fingerprint(state: &GameState) -> u64 {
40    let canonical = CanonicalStateForHash::from_state(state);
41    hash_postcard(&canonical)
42}

44/// Compute a stable fingerprint for an event stream.
45pub fn events_fingerprint(events: &[Event]) -> u64 {
46    hash_postcard(events)
