7/// Check whether a card is allowed under the current curriculum and optional explicit allow-list.
8#[inline(always)]
9pub(crate) fn card_set_allowed(
10    card: &CardStatic,
11    curriculum: &CurriculumConfig,
12    allowed_card_sets: Option<&HashSet<String>>,
13) -> bool {
14    if let Some(set) = allowed_card_sets.or(curriculum.allowed_card_sets_cache.as_ref()) {
15        return match &card.card_set {
16            Some(set_id) => set.contains(set_id),
17            None => false,
18        };
19    }
20    if curriculum.allowed_card_sets.is_empty() {
21        true
22    } else {
23        card.card_set
24            .as_ref()
25            .map(|s| curriculum.allowed_card_sets.iter().any(|a| a == s))
26            .unwrap_or(false)
27    }
28}

30/// Check the level requirement for playing a card from hand.
31#[inline(always)]
32pub(crate) fn meets_level_requirement(
33    card: &CardStatic,
34    level_count: usize,
35    level_delta: i32,
36) -> bool {
37    let required_level = (card.level as i32).saturating_add(level_delta).max(0) as usize;
38    required_level <= level_count
39}

41/// Check the stock cost requirement for playing a card from hand.
42#[inline(always)]
43pub(crate) fn meets_cost_requirement(
44    card: &CardStatic,
45    stock_count: usize,
46    enforce_cost_requirement: bool,
47) -> bool {
48    !enforce_cost_requirement || stock_count >= card.cost as usize
49}

51/// Check the color requirement for playing a card from hand.
52#[inline(always)]
53pub(crate) fn meets_color_requirement(
54    card: &CardStatic,
55    player: &PlayerState,
56    db: &CardDb,
57    enforce_color_requirement: bool,
58    ignore_color_requirement: bool,
59) -> bool {
60    if ignore_color_requirement || !enforce_color_requirement {
61        return true;
62    }
63    if card.level == 0 || card.color == CardColor::Colorless {
64        return true;
65    }
66    for card_id in player.level.iter().chain(player.clock.iter()) {
67        if let Some(c) = db.get(card_id.id) {
68            if c.color == card.color {
69                return true;
70            }
71        }
72    }
73    false
74}

76/// Check all requirements for playing a card from hand under the current curriculum.
77#[inline(always)]
78pub(crate) fn meets_play_requirements(
79    card: &CardStatic,
80    player: &PlayerState,
81    db: &CardDb,
82    curriculum: &CurriculumConfig,
83    level_delta: i32,
84    ignore_color_requirement: bool,
85) -> bool {
86    meets_level_requirement(card, player.level.len(), level_delta)
87        && meets_cost_requirement(
88            card,
89            player.stock.len(),
90            curriculum.enforce_cost_requirement,
91        )
92        && meets_color_requirement(
93            card,
94            player,
95            db,
96            curriculum.enforce_color_requirement,
97            ignore_color_requirement,
98        )
99}
